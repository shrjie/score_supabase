// ---------------- 基礎：試算表 ----------------
function getSS() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (ss) return ss;
  } catch (e) { /* 非綁定執行時會拋錯，改用備援 ID */ }
  return SpreadsheetApp.openById(FALLBACK_SPREADSHEET_ID);
}

function isReserved(name) {
  return name === ACCOUNT_SHEET || name === CONFIG_SHEET || name === AUDIT_SHEET;
}

function listExamNames() {
  var ss = getSS();
  return ss.getSheets().map(function (s) { return s.getName(); })
    .filter(function (n) { return !isReserved(n); });
}
function orderedExamNames() { return listExamNames(); }

function nextExamName(name) {
  var ids = orderedExamNames();
  var i = ids.indexOf(name);
  return (i >= 0 && i + 1 < ids.length) ? ids[i + 1] : null;
}

// ---------------- 設定（含系統標題、老師密碼） ----------------
function ensureConfigSheet() {
  var ss = getSS();
  var sh = ss.getSheetByName(CONFIG_SHEET);
  if (!sh) {
    sh = ss.insertSheet(CONFIG_SHEET);
    sh.getRange(1, 1, 1, 2).setValues([['鍵', '值']]);
    sh.getRange(2, 1, 3, 2).setValues([
      ['SYSTEM_TITLE', DEFAULT_TITLE],
      ['TEACHER_PASSWORD', DEFAULT_TEACHER_PW],
      ['CLASS_NO', DEFAULT_CLASS_NO]
    ]);
  }
  return sh;
}

function getConfig(key, defVal) {
  var sh = ensureConfigSheet();
  var data = sh.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === key) return String(data[i][1]);
  }
  return defVal;
}

function setConfig(key, val) {
  var sh = ensureConfigSheet();
  var data = sh.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === key) {
      sh.getRange(i + 1, 2).setValue(val);
      return;
    }
  }
  sh.appendRow([key, val]);
}

function getSystemTitle() {
  var t = getConfig('SYSTEM_TITLE', DEFAULT_TITLE);
  return t ? t : DEFAULT_TITLE;
}

function verifyTeacher(pw) {
  return String(pw || '') === getConfig('TEACHER_PASSWORD', DEFAULT_TEACHER_PW);
}

// ---------------- 帳號 ----------------
function isTruthy(v) {
  if (v === true) return true;
  var s = String(v == null ? '' : v).trim();
  return s === '1' || s.toUpperCase() === 'TRUE' || s.indexOf('測') >= 0 || s === '🧪';
}

function normSeat(s) {
  var t = String(s == null ? '' : s).trim();
  if (t === '') return '';
  var n = Number(t);
  if (!isNaN(n)) return String(n);
  return t;
}

// 讀學生帳密表：A帳號 B密碼 C座號 D姓名 E班級(選) F測試(選)
function readAccounts() {
  var ss = getSS();
  var sh = ss.getSheetByName(ACCOUNT_SHEET);
  if (!sh) throw new Error('找不到「' + ACCOUNT_SHEET + '」工作表');
  var data = sh.getDataRange().getValues();
  var width = data.length ? data[0].length : 0;
  var out = [];
  for (var i = 1; i < data.length; i++) {
    var acc = String(data[i][0] == null ? '' : data[i][0]).trim();
    if (!acc) continue;
    out.push({
      account: acc,
      password: String(data[i][1] == null ? '' : data[i][1]).trim(),
      seat: String(data[i][2] == null ? '' : data[i][2]).trim(),
      name: String(data[i][3] == null ? '' : data[i][3]).trim(),
      classNo: width > 4 ? (String(data[i][4] == null ? '' : data[i][4]).trim() || DEFAULT_CLASS_NO) : DEFAULT_CLASS_NO,
      excluded: width > 5 ? isTruthy(data[i][5]) : false,
      row: i + 1
    });
  }
  return out;
}

function findSeatCol(headers) {
  for (var j = 0; j < headers.length; j++) {
    if (SEAT_COL_NAMES.indexOf(headers[j]) >= 0) return j;
  }
  return 0;
}
function findNameCol(headers) {
  for (var j = 0; j < headers.length; j++) {
    if (NAME_COL_NAMES.indexOf(headers[j]) >= 0) return j;
  }
  return 1;
}

// ---------------- 計分引擎（對齊 D1 recalcChain） ----------------
function parseWeight(name, fallback) {
  var fb = (fallback == null ? 1 : fallback);
  var s = String(name || '');
  var star = s.lastIndexOf('*');
  if (star < 0) return fb;
  var w = parseFloat(s.substring(star + 1));
  return (w > 0) ? w : fb;
}
function stripWeightSuffix(name) {
  var s = String(name || '').trim();
  var star = s.lastIndexOf('*');
  if (star > 0) {
    var tail = s.substring(star + 1).trim();
    if (tail !== '' && !isNaN(Number(tail))) return s.substring(0, star).trim();
  }
  return s;
}
function syncWeightSuffix(name, weight) {
  var base = stripWeightSuffix(name) || String(name || '').trim();
  var w = (weight === Math.round(weight)) ? String(Math.round(weight)) : String(weight);
  return base + '*' + w;
}
function isPureSubject(h) {
  return !!h && !DERIVED_RE.test(h);
}
function isGradeCol(headers, idx, seatCol, nameCol) {
  if (idx === seatCol || idx === nameCol) return false;
  var h = headers[idx];
  return !!h && !DERIVED_RE.test(h);
}
function validScore(v) {
  if (v === null || v === undefined || v === '') return { ok: true, value: null };
  var n = Number(v);
  if (!isFinite(n)) return { ok: false, msg: '成績必須是數字，收到：' + v };
  if (n < -10000 || n > 10000) return { ok: false, msg: '成績超出範圍：' + v };
  return { ok: true, value: Math.round(n * 1000000) / 1000000 };
}
// Excel RANK.EQ：由高到低，同分並列、下一名跳號
function compRank(entries) {
  var sorted = entries.slice().sort(function (a, b) { return b.value - a.value; });
  var ranks = {};
  var prev = null, prevRank = 0;
  for (var i = 0; i < sorted.length; i++) {
    var r = (sorted[i].value === prev) ? prevRank : (i + 1);
    ranks[sorted[i].key] = r;
    prev = sorted[i].value; prevRank = r;
  }
  return ranks;
}
function toNum(v) {
  if (v === null || v === undefined || v === '') return null;
  var n = Number(v);
  return isFinite(n) ? n : null;
}

// 確保衍生欄存在（總分/平均/班名次/前名次/進退步），回傳最新 headers
function ensureDerivedColumns(sh, headers) {
  var last = headers.length;
  for (var k = 0; k < DERIVED_COLS.length; k++) {
    if (headers.indexOf(DERIVED_COLS[k]) < 0) {
      sh.getRange(1, last + 1).setValue(DERIVED_COLS[k]);
      headers.push(DERIVED_COLS[k]);
      last++;
    }
  }
  return headers;
}

// 重算單次考試；prevClass：{normSeat: 上次班名次}
// 回傳 {cells, classRanks}
function recalcOneExam(sheetName, prevClass) {
  var ss = getSS();
  var sh = ss.getSheetByName(sheetName);
  if (!sh) throw new Error('找不到工作表：' + sheetName);
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return { cells: 0, classRanks: {} };
  var headers = data[0].map(function (h) { return String(h == null ? '' : h).trim(); });
  headers = ensureDerivedColumns(sh, headers);
  var W = headers.length;
  var seatCol = findSeatCol(headers);
  var nameCol = findNameCol(headers);

  var accounts = readAccounts();
  var exclMap = {};
  accounts.forEach(function (a) { if (a.excluded) exclMap[normSeat(a.seat)] = true; });

  var pureCols = [];
  for (var c = 0; c < headers.length; c++) {
    if (isGradeCol(headers, c, seatCol, nameCol)) pureCols.push(c);
  }
  var weights = {};
  pureCols.forEach(function (c) { weights[c] = parseWeight(headers[c], 1); });

  // 科排名欄 ↔ 純科目配對（國排名→國文*5，依首字）
  var rankPairs = [];
  for (var r2 = 0; r2 < headers.length; r2++) {
    var h = headers[r2];
    if (!/排名|名次/.test(h)) continue;
    if (h === '班名次' || h === '前名次') continue;
    var base = h.replace(/排名|名次/g, '');
    for (var p = 0; p < pureCols.length; p++) {
      if (stripWeightSuffix(headers[pureCols[p]]).indexOf(base) === 0) {
        rankPairs.push({ rank: r2, subj: pureCols[p] });
        break;
      }
    }
  }
  var colTotal = headers.indexOf('總分');
  var colAvg = -1, colClass = headers.indexOf('班名次'),
      colPrev = headers.indexOf('前名次'), colProg = headers.indexOf('進退步');
  for (var a2 = 0; a2 < headers.length; a2++) {
    if (headers[a2].indexOf('平均') >= 0) { colAvg = a2; break; }
  }

  // 補齊列寬
  for (var i = 1; i < data.length; i++) {
    while (data[i].length < W) data[i].push('');
  }
  var seatKey = function (rowIdx) { return normSeat(data[rowIdx][seatCol]); };

  // 總分 / 平均
  var totals = {};
  for (var r3 = 1; r3 < data.length; r3++) {
    var t = 0, w = 0, has = false;
    for (var q = 0; q < pureCols.length; q++) {
      var v = toNum(data[r3][pureCols[q]]);
      if (v !== null) { t += v * weights[pureCols[q]]; w += weights[pureCols[q]]; has = true; }
    }
    var tv = has ? Math.round(t * 10000) / 10000 : null;
    totals[seatKey(r3)] = tv;
    if (colTotal >= 0) data[r3][colTotal] = tv;
    if (colAvg >= 0) data[r3][colAvg] = (has && w) ? Math.round((t / w) * 100000000) / 100000000 : null;
  }
  // 科排名（測試帳不列入，留白）
  rankPairs.forEach(function (rp) {
    var entries = [];
    for (var r4 = 1; r4 < data.length; r4++) {
      var k = seatKey(r4);
      if (!String(data[r4][seatCol]).trim()) continue;
      if (exclMap[k]) continue;
      var v = toNum(data[r4][rp.subj]);
      if (v !== null) entries.push({ key: k + '_' + r4, value: v });
    }
    var ranks = compRank(entries);
    for (var r5 = 1; r5 < data.length; r5++) {
      var kk = seatKey(r5);
      var val = null;
      if (!exclMap[kk] && String(data[r5][seatCol]).trim()) {
        var hit = ranks[kk + '_' + r5];
        val = (hit == null ? null : hit);
      }
      data[r5][rp.rank] = val;
    }
  });
  // 班名次（測試帳不列入）
  var classEntries = [];
  for (var r6 = 1; r6 < data.length; r6++) {
    var k6 = seatKey(r6);
    if (!String(data[r6][seatCol]).trim()) continue;
    if (exclMap[k6]) continue;
    var tv6 = totals[k6];
    if (tv6 !== null && tv6 !== undefined) classEntries.push({ key: k6 + '_' + r6, value: tv6 });
  }
  var classRanksArr = compRank(classEntries);
  var classRanks = {};
  for (var r7 = 1; r7 < data.length; r7++) {
    var k7 = seatKey(r7);
    var cv = null;
    if (!exclMap[k7] && String(data[r7][seatCol]).trim()) {
      var h7 = classRanksArr[k7 + '_' + r7];
      cv = (h7 == null ? null : h7);
      if (cv !== null) classRanks[k7] = cv;
    }
    if (colClass >= 0) data[r7][colClass] = cv;
  }
  // 前名次 / 進退步
  for (var r8 = 1; r8 < data.length; r8++) {
    var k8 = seatKey(r8);
    var prev = (prevClass && prevClass[k8] != null) ? prevClass[k8] : null;
    var cur = (classRanks[k8] != null) ? classRanks[k8] : null;
    if (colPrev >= 0) data[r8][colPrev] = prev;
    if (colProg >= 0) data[r8][colProg] = (prev !== null && cur !== null) ? prev - cur : null;
  }
  // 寫回（只寫衍生欄）
  var derived = [];
  if (colTotal >= 0) derived.push(colTotal);
  if (colAvg >= 0) derived.push(colAvg);
  if (colClass >= 0) derived.push(colClass);
  if (colPrev >= 0) derived.push(colPrev);
  if (colProg >= 0) derived.push(colProg);
  rankPairs.forEach(function (rp) { derived.push(rp.rank); });
  derived.forEach(function (c) {
    var col = [];
    for (var r9 = 1; r9 < data.length; r9++) {
      var v9 = data[r9][c];
      col.push([v9 === null ? '' : v9]);
    }
    sh.getRange(2, c + 1, data.length - 1, 1).setValues(col);
  });
  return { cells: derived.length * (data.length - 1), classRanks: classRanks };
}

function recalcChain(names, seedPrev) {
  var list = names || orderedExamNames();
  var prev = seedPrev || {};
  var cells = 0;
  list.forEach(function (n) {
    var r = recalcOneExam(n, prev);
    prev = r.classRanks;
    cells += r.cells;
  });
  return { exams: list.length, cells: cells };
}

// 讀取某次考試已存的班名次 {normSeat: rank}（部分重算的種子，不寫回該表）
function readClassRanks(sheetName) {
  var out = {};
  try {
    var sh = getSS().getSheetByName(String(sheetName).trim());
    if (!sh) return out;
    var data = sh.getDataRange().getValues();
    if (data.length < 2) return out;
    var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
    var ci = headers.indexOf('班名次');
    if (ci < 0) return out;
    var seatCol = findSeatCol(headers);
    for (var i = 1; i < data.length; i++) {
      var k = normSeat(data[i][seatCol]);
      var v = toNum(data[i][ci]);
      if (k && v !== null) out[k] = v;
    }
  } catch (e) { /* 讀不到就當無種子 */ }
  return out;
}

// 單次異動重算窗口：[前一次, 本次, 下一次]，並以前前次班名次當種子，
// 使本次的前名次/進退步不被清空，且下一次的前名次/進退步跟著最新分數連動
function recalcWindow(centerExam) {
  var order = orderedExamNames();
  var c = String(centerExam).trim();
  var i = order.indexOf(c);
  if (i < 0) return recalcChain([c]);
  var start = Math.max(0, i - 1);
  var names = order.slice(start, i + 2);
  var seed = (start > 0) ? readClassRanks(order[start - 1]) : {};
  return recalcChain(names, seed);
}

// ---------------- API 入口 ----------------
function doGet(e) {
  var params = (e && e.parameter) ? e.parameter : {};
  if (params.action) return handleAPI(params);
  // 列印專用頁（新分頁開啟，頂層文件列印，不受嵌入 iframe 影響）
  if (params.print === 'slip') return printSlipPage(params.u, params.e);
  if (params.print === 'report') return printReportPage(params.e, params.pw);
  if (params.print === 'slips') return printSlipsPage(params.e, params.pw);
  // 單頁版：學生＋老師同一頁，站內按鈕切換（Google Sites 嵌入 iframe 跳轉會被擋）
  return HtmlService.createHtmlOutput(getAppHTML(params.page === 'teacher'))
    .setTitle(getSystemTitle())
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  return handleAPI((e && e.parameter) ? e.parameter : {});
}

function handleAPI(params) {
  var result;
  try {
    result = apiCall(params.action || '', params);
  } catch (err) {
    result = { success: false, message: '伺服器錯誤：' + err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// google.script.run 用：apiCall(action, params)
function apiCall(action, params) {
  params = params || {};
  try {
    if (action === 'login') return hLogin(params.username, params.password);
    if (action === 'scores') return hStudentScores(params.username, params.sheet || params.exam);
    if (action === 'ping') return { success: true, message: '後端連線正常' };
    if (action === 'teacher_login') return hTeacherLogin(params.password);
    if (action === 'teacher_change_password') return hChangePassword(params.old_password, params.new_password);
    if (action === 'config_get_title') return { success: true, title: getSystemTitle() };
    if (action === 'config_set_title') return hConfigSetTitle(params.tpw, params.title);
    if (action === 'exams_list') return { success: true, exams: listExamNames(), title: getSystemTitle() };
    if (action === 'exam_create') return hExamCreate(params.tpw, params.name, params.copy_from);
    if (action === 'exam_rename') return hExamRename(params.tpw, params.oldName || params.old_name, params.newName || params.new_name);
    if (action === 'exam_delete') return hExamDelete(params.tpw, params.name, params.password);
    if (action === 'subjects_list') return hSubjectsList(params.exam);
    if (action === 'subject_create') return hSubjectCreate(params.tpw, params.code, params.name, params.weight);
    if (action === 'subject_edit') return hSubjectEdit(params.tpw, params.oldName || params.old_name, params.name, params.weight);
    if (action === 'grid') return hGrid(params.exam);
    if (action === 'score_set') return hScoreSet(params.tpw, params.exam, params.student_id, params.subject_id, params.score);
    if (action === 'score_bulk') return hScoreBulk(params.tpw, params.exam, params.cells, params.rows);
    if (action === 'student_scores') return hStudentScores(params.username, params.exam);
    if (action === 'student_trend') return hStudentTrend(params.username);
    if (action === 'student_stats') return hStudentStats(params.exam);
    if (action === 'students_list') return hStudentsList(params.tpw);
    if (action === 'student_upsert') return hStudentUpsert(params.tpw, params);
    if (action === 'audit_list') return hAuditList(params.tpw, params.limit);
    if (action === 'recalc') return hRecalc(params.tpw, params.exam);
    if (action === 'recalc_all') return hRecalc(params.tpw, null);
    return { success: false, message: '未知 action：' + action };
  } catch (err) {
    return { success: false, message: '伺服器錯誤：' + err.message };
  }
}

// ---------------- 登入 ----------------
function hLogin(username, password) {
  if (!username || !password) return { success: false, message: '帳號或密碼不可為空白' };
  var accounts = readAccounts();
  var found = null;
  for (var i = 0; i < accounts.length; i++) {
    if (accounts[i].account === String(username).trim()) { found = accounts[i]; break; }
  }
  if (!found) return { success: false, message: '無此帳號，請確認後再試' };
  if (found.password !== String(password).trim()) return { success: false, message: '密碼錯誤，請再試一次' };
  return {
    success: true,
    user: { username: found.account, seat: found.seat, name: found.name, class_no: found.classNo },
    sheets: listExamNames(),
    title: getSystemTitle()
  };
}

function hTeacherLogin(password) {
  if (!password) return { success: false, message: '請輸入管理密碼' };
  if (!verifyTeacher(password)) return { success: false, message: '管理密碼錯誤' };
  return { success: true, title: getSystemTitle() };
}

function hChangePassword(oldPw, newPw) {
  if (!verifyTeacher(oldPw)) return { success: false, message: '舊密碼錯誤' };
  if (!newPw || String(newPw).length < 6) return { success: false, message: '新密碼至少 6 碼' };
  setConfig('TEACHER_PASSWORD', String(newPw));
  logAudit('teacher', 'change_password', '管理密碼已變更');
  return { success: true, message: '密碼已變更' };
}

function hConfigSetTitle(tpw, title) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var t = String(title || '').trim();
  if (!t) return { success: false, message: '標題不可空白' };
  setConfig('SYSTEM_TITLE', t);
  logAudit('teacher', 'edit_title', '系統標題改為：' + t);
  return { success: true, title: t };
}

// ---------------- 考試 ----------------
function hExamCreate(tpw, name, copyFrom) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var nm = String(name || '').trim();
  if (!nm) return { success: false, message: '考試名稱不可空白' };
  if (isReserved(nm)) return { success: false, message: '名稱保留，請換一個' };
  var ss = getSS();
  if (ss.getSheetByName(nm)) return { success: false, message: '考試名稱重複' };
  var sh = ss.insertSheet(nm);
  if (copyFrom && ss.getSheetByName(copyFrom)) {
    var src = ss.getSheetByName(copyFrom);
    var lastCol = src.getLastColumn();
    var head = src.getRange(1, 1, 1, lastCol).getValues()[0];
    sh.getRange(1, 1, 1, head.length).setValues([head]);
  } else {
    sh.getRange(1, 1, 1, 2).setValues([['座號', '姓名']]);
  }
  var r = recalcChain();
  var prevFilled = 0;
  try {
    var d = sh.getDataRange().getValues();
    var h = d[0].map(function (x) { return String(x).trim(); });
    var pi = h.indexOf('前名次');
    if (pi >= 0) {
      for (var i = 1; i < d.length; i++) {
        if (d[i][pi] !== '' && d[i][pi] !== null) prevFilled++;
      }
    }
  } catch (e) { /* 忽略 */ }
  logAudit('teacher', 'create_exam', '新增考試：' + nm + ' prev_filled=' + prevFilled);
  return { success: true, exam: nm, prev_filled: prevFilled, recalc: r };
}

function hSubjectCreate(tpw, code, name, weight) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var nm = String(name || code || '').trim();
  if (!nm) return { success: false, message: '科目名稱不可空白' };
  var w = (weight !== undefined && weight !== null && weight !== '') ? Number(weight) : parseWeight(nm, 1);
  if (!isFinite(w) || w <= 0) return { success: false, message: '權重必須是正數' };
  var finalName = isPureSubject(nm) ? syncWeightSuffix(nm, w) : nm;
  var exams = orderedExamNames();
  exams.forEach(function (n) {
    var sh = getSS().getSheetByName(n);
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
      .map(function (x) { return String(x == null ? '' : x).trim(); });
    if (head.indexOf(finalName) >= 0) return;
    var at = -1;
    for (var j = 0; j < head.length; j++) {
      if (DERIVED_RE.test(head[j])) { at = j; break; }
    }
    if (at >= 0) {
      sh.insertColumnBefore(at + 1);
      sh.getRange(1, at + 1).setValue(finalName);
    } else {
      sh.getRange(1, head.length + 1).setValue(finalName);
    }
  });
  var r = recalcChain();
  logAudit('teacher', 'create_subject', '新增科目：' + finalName + ' 權重=' + w);
  return { success: true, subject: finalName, recalc: r };
}

function hSubjectEdit(tpw, oldName, newName, weight) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var o = String(oldName || '').trim();
  var nm = String(newName == null ? o : newName).trim();
  if (!o || !nm) return { success: false, message: '科目名稱不可空白' };
  var exams = orderedExamNames();
  var oldW = 1, foundAny = false;
  exams.forEach(function (n) {
    var sh = getSS().getSheetByName(n);
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
      .map(function (x) { return String(x == null ? '' : x).trim(); });
    if (head.indexOf(o) >= 0) { foundAny = true; oldW = parseWeight(o, 1); }
  });
  if (!foundAny) return { success: false, message: '找不到科目：' + o };
  var w;
  if (weight !== undefined && weight !== null && weight !== '') w = Number(weight);
  else if (nm !== o) w = parseWeight(nm, oldW);
  else w = oldW;
  if (!isFinite(w) || w <= 0) return { success: false, message: '權重必須是正數' };
  var finalName = isPureSubject(nm) ? syncWeightSuffix(nm, w) : nm;
  var dup = false;
  exams.forEach(function (n) {
    var sh = getSS().getSheetByName(n);
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
      .map(function (x) { return String(x == null ? '' : x).trim(); });
    var idx = head.indexOf(o);
    if (idx >= 0) {
      if (finalName !== o && head.indexOf(finalName) >= 0) { dup = true; return; }
      sh.getRange(1, idx + 1).setValue(finalName);
    }
  });
  if (dup) return { success: false, message: '名稱重複：' + finalName };
  var r = recalcChain();
  logAudit('teacher', 'edit_subject', o + ' → ' + finalName + ' 權重=' + w);
  return { success: true, name: finalName, weight: w, recalc: r };
}

// ---------------- 老師：全班矩陣 / 單格 / 整批 ----------------
function hGrid(exam) {
  var nm = String(exam || '').trim();
  var sh = getSS().getSheetByName(nm);
  if (!sh) return { success: false, message: '找不到考試：' + nm };
  var data = sh.getDataRange().getValues();
  if (data.length < 1) return { success: false, message: '工作表無資料' };
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers), nameCol = findNameCol(headers);
  var rowBySeat = {};
  for (var i = 1; i < data.length; i++) {
    var k = normSeat(data[i][seatCol]);
    if (k && rowBySeat[k] == null) rowBySeat[k] = i;
  }
  var accounts = readAccounts();
  accounts.sort(function (a, b) {
    var x = parseFloat(a.seat), y = parseFloat(b.seat);
    if (!isNaN(x) && !isNaN(y)) return x - y;
    return String(a.seat) < String(b.seat) ? -1 : 1;
  });
  var students = accounts.map(function (a) {
    return { id: a.seat, account: a.account, seat: a.seat, name: a.name, class_no: a.classNo, excluded: a.excluded };
  });
  var subjects = [];
  headers.forEach(function (h) {
    if (!h) return;
    var idx = headers.indexOf(h);
    if (idx === seatCol || idx === nameCol) return;
    subjects.push({ id: h, code: h, name: h, weight: parseWeight(h, 1) });
  });
  var scores = {};
  accounts.forEach(function (a) {
    var ri = rowBySeat[normSeat(a.seat)];
    subjects.forEach(function (s) {
      var ci = headers.indexOf(s.id);
      var v = (ri != null) ? data[ri][ci] : '';
      scores[a.seat + '_' + s.id] = (v === '' || v == null) ? '' : v;
    });
  });
  return { success: true, students: students, subjects: subjects, scores: scores };
}

function findRowBySeat(data, seatCol, seat) {
  var key = normSeat(seat);
  for (var i = 1; i < data.length; i++) {
    if (normSeat(data[i][seatCol]) === key) return i;
  }
  return -1;
}

function hScoreSet(tpw, exam, studentId, subjectId, score) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var sh = getSS().getSheetByName(String(exam || '').trim());
  if (!sh) return { success: false, message: '找不到考試' };
  var v = validScore(score);
  if (!v.ok) return { success: false, message: v.msg };
  var data = sh.getDataRange().getValues();
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers);
  var ci = headers.indexOf(String(subjectId || '').trim());
  if (ci < 0) return { success: false, message: '找不到科目：' + subjectId };
  if (DERIVED_RE.test(headers[ci])) return { success: false, message: '自動計算欄不可手改' };
  var ri = findRowBySeat(data, seatCol, studentId);
  if (ri < 0) return { success: false, message: '找不到座號：' + studentId };
  if (v.value === null) sh.getRange(ri + 1, ci + 1).clearContent();
  else sh.getRange(ri + 1, ci + 1).setValue(v.value);
  logAudit('teacher', 'edit_cell', '考試=' + exam + ' 座號=' + studentId + ' 科目=' + subjectId + ' 分數=' + v.value);
  var r = recalcWindow(exam);
  return { success: true, score: v.value, recalc: r };
}

function hScoreBulk(tpw, exam, cells, rows) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var nm = String(exam || '').trim();
  var sh = getSS().getSheetByName(nm);
  if (!sh) return { success: false, message: '找不到考試' };
  var data = sh.getDataRange().getValues();
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers);
  var errors = [], count = 0;
  var apply = function (seat, subject, score) {
    var ci = headers.indexOf(String(subject).trim());
    if (ci < 0) { errors.push('未知科目：' + subject); return; }
    if (DERIVED_RE.test(headers[ci])) return; // 自動欄跳過，由重算填入
    var ri = findRowBySeat(data, seatCol, seat);
    if (ri < 0) { errors.push('座號不在名單：' + seat); return; }
    var v = validScore(score);
    if (!v.ok) { errors.push('座號=' + seat + ' 科目=' + subject + '：' + v.msg); return; }
    data[ri][ci] = (v.value === null ? '' : v.value);
    count++;
  };
  (cells || []).forEach(function (c) { apply(c.student_id, c.subject_id, c.score); });
  (rows || []).forEach(function (r) {
    if (!r || !r.scores) return;
    for (var k in r.scores) {
      if (Object.prototype.hasOwnProperty.call(r.scores, k)) apply(r.student_id, k, r.scores[k]);
    }
  });
  if (!count) return { success: false, message: '無有效資料', errors: errors };
  sh.getRange(1, 1, data.length, headers.length).setValues(data);
  logAudit('teacher', 'bulk_upsert', '考試=' + nm + ' 寫入' + count + '格');
  var r2 = recalcWindow(nm);
  return { success: true, updated: count, errors: errors, recalc: r2 };
}

function findStudentRowInSheet(sh, seat) {
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return { data: data, headers: [], rowIdx: -1 };
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers);
  var key = normSeat(seat);
  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (normSeat(data[i][seatCol]) === key) { rowIdx = i; break; }
  }
  return { data: data, headers: headers, rowIdx: rowIdx, seatCol: seatCol };
}

function hStudentScores(username, examName) {
  if (!username || !examName) return { success: false, message: '參數不完整' };
  var accounts = readAccounts();
  var me = null;
  for (var i = 0; i < accounts.length; i++) {
    if (accounts[i].account === String(username).trim()) { me = accounts[i]; break; }
  }
  if (!me) return { success: false, message: '找不到此帳號' };
  var sh = getSS().getSheetByName(String(examName).trim());
  if (!sh) return { success: false, message: '找不到考試：' + examName };
  var f = findStudentRowInSheet(sh, me.seat);
  if (f.rowIdx < 0) return { success: false, message: '找不到座號「' + me.seat + '」的成績資料' };
  var headers = f.headers, row = f.data[f.rowIdx];
  var formattedRow = row.map(function (val, idx) {
    if (typeof val === 'number') {
      var h = headers[idx] || '';
      if (h.indexOf('平均') >= 0) return val.toFixed(2);
      return val;
    }
    return String(val == null ? '' : val);
  });
  var scores = headers.map(function (h, idx) {
    var raw = row[idx];
    var val = (raw === '' || raw == null) ? null : raw;
    return { subject: h, score: val };
  });
  var avgIdx = -1;
  for (var a = 0; a < headers.length; a++) {
    if (headers[a].indexOf('平均') >= 0) { avgIdx = a; break; }
  }
  var official = (avgIdx >= 0) ? toNum(row[avgIdx]) : null;
  var sum = 0, n = 0;
  var seatCol = findSeatCol(headers), nameCol = findNameCol(headers);
  headers.forEach(function (h, idx) {
    if (idx === seatCol || idx === nameCol) return;
    if (!isPureSubject(h)) return;
    var v = toNum(row[idx]);
    if (v !== null) { sum += v; n++; }
  });
  var avg = n ? Math.round((sum / n) * 100) / 100 : null;
  // 結構化：各科（分數＋配對排名）與總結（總分/平均/班名次/前名次/進退步）分開
  var cellVal = function (idx) {
    var raw = row[idx];
    return (raw === '' || raw == null) ? null : raw;
  };
  var rankIdxFor = function (pi) {
    var subj = stripWeightSuffix(headers[pi]);
    for (var j = 0; j < headers.length; j++) {
      var h = headers[j];
      if (!/排名|名次/.test(h)) continue;
      if (h === '班名次' || h === '前名次') continue;
      var base = h.replace(/排名|名次/g, '');
      if (!base) continue;
      if (subj.indexOf(base) === 0) return j;
    }
    return -1;
  };
  var subjRows = [];
  headers.forEach(function (h, idx) {
    if (idx === seatCol || idx === nameCol || !h) return;
    if (!isPureSubject(h)) return;
    var ri = rankIdxFor(idx);
    subjRows.push({ name: stripWeightSuffix(h), full: h, score: cellVal(idx), rank: ri >= 0 ? cellVal(ri) : null });
  });
  var summaryRows = [];
  headers.forEach(function (h, idx) {
    if (idx === seatCol || idx === nameCol || !h) return;
    if (isPureSubject(h)) return;
    if (/排名|名次/.test(h) && h !== '班名次' && h !== '前名次') return;
    summaryRows.push({ name: h, value: cellVal(idx) });
  });
  return { success: true, exam: String(examName).trim(), headers: headers, row: formattedRow, scores: scores, average: avg, official_average: official, subjects: subjRows, summary: summaryRows };
}

function hStudentStats(exam) {
  var nm = String(exam || '').trim();
  var sh = getSS().getSheetByName(nm);
  if (!sh) return { success: false, message: '找不到考試' };
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return { success: true, stats: [] };
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers), nameCol = findNameCol(headers);
  var accounts = readAccounts();
  var kidSet = {};
  accounts.forEach(function (a) { if (!a.excluded) kidSet[normSeat(a.seat)] = true; });
  var stats = [];
  for (var c = 0; c < headers.length; c++) {
    if (c === seatCol || c === nameCol) continue;
    if (!isPureSubject(headers[c])) continue;
    var vals = [];
    for (var i = 1; i < data.length; i++) {
      if (!kidSet[normSeat(data[i][seatCol])]) continue;
      var v = toNum(data[i][c]);
      if (v !== null) vals.push(v);
    }
    var nmShort = stripWeightSuffix(headers[c]);
    if (!vals.length) { stats.push({ name: nmShort, n: 0 }); continue; }
    var sum = vals.reduce(function (a, b) { return a + b; }, 0);
    var pass = vals.filter(function (x) { return x >= 60; }).length;
    stats.push({
      name: nmShort, n: vals.length,
      avg: Math.round(sum / vals.length * 100) / 100,
      max: Math.max.apply(null, vals), min: Math.min.apply(null, vals),
      rate: Math.round(pass / vals.length * 1000) / 10,
      g90: vals.filter(function (x) { return x >= 90; }).length,
      g80: vals.filter(function (x) { return x >= 80 && x < 90; }).length,
      g70: vals.filter(function (x) { return x >= 70 && x < 80; }).length,
      g60: vals.filter(function (x) { return x >= 60 && x < 70; }).length,
      g0: vals.filter(function (x) { return x < 60; }).length
    });
  }
  return { success: true, stats: stats };
}

function hStudentUpsert(tpw, p) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var account = String(p.account || '').trim();
  var seat = String(p.seat || '').trim();
  var name = String(p.name || '').trim();
  if (!account || !seat || !name) return { success: false, message: '帳號/座號/姓名不可空白' };
  var classNo = String(p.class_no || getConfig('CLASS_NO', DEFAULT_CLASS_NO)).trim() || DEFAULT_CLASS_NO;
  var excl = p.excluded ? 1 : 0;
  var ss = getSS();
  var sh = ss.getSheetByName(ACCOUNT_SHEET);
  var data = sh.getDataRange().getValues();
  if (data[0].length < 5) sh.getRange(1, 5).setValue('班級');
  if (data[0].length < 6) sh.getRange(1, 6).setValue('測試');
  var ri = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === account) { ri = i + 1; break; }
  }
  if (ri < 0) {
    if (!p.password) return { success: false, message: '新增學生需提供初始密碼' };
    sh.appendRow([account, String(p.password), seat, name, classNo, excl]);
  } else {
    if (p.password) sh.getRange(ri, 2).setValue(String(p.password));
    sh.getRange(ri, 3, 1, 4).setValues([[seat, name, classNo, excl]]);
  }
  logAudit('teacher', 'upsert_student', '帳號=' + account + (p.password ? '（重設密碼）' : '') + ' 測試=' + excl);
  var r = recalcChain();
  return { success: true, recalc: r };
}

function hAuditList(tpw, limit) {
  if (!verifyTeacher(tpw)) return { success: false, message: '需要老師權限' };
  var lim = parseInt(limit || 200, 10);
  if (!isFinite(lim) || lim < 1) lim = 200;
  if (lim > 500) lim = 500;
  var sh = ensureAuditSheet();
  var data = sh.getDataRange().getValues();
  var logs = [];
  for (var i = data.length - 1; i >= 1 && logs.length < lim; i--) {
    var t = data[i][0] instanceof Date ? Math.floor(data[i][0].getTime() / 1000) : 0;
    logs.push({ created_at: t, teacher: String(data[i][1] || ''), action: String(data[i][2] || ''), detail: String(data[i][3] || '') });
  }
  return { success: true, logs: logs };
}

    label = orderedExamNames().join(',');
  }
  logAudit('teacher', 'recalc', '考試=' + label);

}

function drawTrend(pts) {
  var cv = document.getElementById('trend-cv');
  var pw = cv.parentElement.clientWidth || 640;
  var W = Math.max(pw - 8, pts.length * 88 + 68, 280), H = 260, dpr = window.devicePixelRatio || 1;
  cv.width = W * dpr; cv.height = H * dpr; cv.style.width = W + 'px'; cv.style.height = H + 'px';
  var c = cv.getContext('2d'); c.setTransform(dpr, 0, 0, dpr, 0, 0); c.clearRect(0, 0, W, H);
  var padL = 36, padR = 32, padT = 16, padB = 30;
  var avgs = pts.map(function (p) { return p.average; }).filter(function (v) { return v !== null && v !== undefined; });
  var ranks = pts.map(function (p) { return p.rank; }).filter(function (v) { return v !== null && v !== undefined; });
  var aMin = avgs.length ? Math.min.apply(null, avgs) : 0;
  var aMax = avgs.length ? Math.max.apply(null, avgs) : 100;
  if (aMin === aMax) { aMin -= 5; aMax += 5; }
  var rMax = ranks.length ? Math.max.apply(null, ranks.concat([1])) : 1;
  var X = function (i) { return padL + (W - padL - padR) * (pts.length === 1 ? 0.5 : i / (pts.length - 1)); };
  var YA = function (v) { return padT + (H - padT - padB) * (1 - (v - aMin) / (aMax - aMin)); };
  var YR = function (v) { return padT + (H - padT - padB) * ((v - 1) / ((rMax - 1) || 1)); };
  c.lineWidth = 1; c.font = '11px sans-serif';
  for (var g = 0; g <= 4; g++) {
    var y = padT + (H - padT - padB) * g / 4;
    c.strokeStyle = '#cfe6d4'; c.beginPath(); c.moveTo(padL, y); c.lineTo(W - padR, y); c.stroke();
    c.fillStyle = '#1565c0'; c.fillText((aMax - (aMax - aMin) * g / 4).toFixed(1), 2, y + 4);
    c.fillStyle = '#d81b60'; c.fillText(String(Math.round(1 + (rMax - 1) * g / 4)), W - padR + 5, y + 4);
  }
  c.strokeStyle = '#1f3a2d'; c.lineWidth = 2;
  c.beginPath(); c.moveTo(padL, padT - 6); c.lineTo(padL, H - padB); c.lineTo(W - padR + 6, H - padB); c.stroke();
  c.beginPath(); c.moveTo(W - padR, padT - 6); c.lineTo(W - padR, H - padB); c.stroke();
  pts.forEach(function (p, i) { var x = X(i); c.beginPath(); c.moveTo(x, H - padB); c.lineTo(x, H - padB + 5); c.stroke(); });
  c.fillStyle = '#3d7a52';
  pts.forEach(function (p, i) { c.fillText(String(p.exam_name || '').slice(0, 5), X(i) - 16, H - 10); });
  c.font = 'bold 11px sans-serif';
  if (avgs.length) { c.fillStyle = '#1565c0'; c.fillText('● 平均', padL + 6, padT + 2); }
  if (ranks.length) { c.fillStyle = '#d81b60'; c.fillText('● 班名次', padL + 56, padT + 2); }
  var line = function (key, Y, color) {
    c.strokeStyle = color; c.lineWidth = 2.5; c.beginPath();
    var started = false;
    pts.forEach(function (p, i) {
      var v = p[key];
      if (v === null || v === undefined) return;
      var x = X(i), y = Y(v);
      if (!started) { c.moveTo(x, y); started = true; } else { c.lineTo(x, y); }
    });
    c.stroke(); c.fillStyle = color;
    pts.forEach(function (p, i) {
      var v = p[key];
      if (v === null || v === undefined) return;
      c.beginPath(); c.arc(X(i), Y(v), 3.5, 0, 7); c.fill();
    });
  };
  if (avgs.length) line('average', YA, '#1565c0');
  if (ranks.length) line('rank', YR, '#d81b60');
}
var lastRadar = null;
function drawRadar(subs) {
  var card = document.getElementById('radar-card');
  var items = (subs || []).map(function (s) {
    var raw = (s.score !== undefined && s.score !== null && s.score !== '') ? s.score : s.value;
    var v = (raw === undefined || raw === null || raw === '') ? null : Number(raw);
    return { name: s.name, value: (v === null || !isFinite(v)) ? null : v };
  }).filter(function (o) { return o.value !== null; });
  if (items.length < 3) { card.style.display = 'none'; lastRadar = null; return; }
  lastRadar = items;
  card.style.display = 'block';
  var cv = document.getElementById('radar-cv');
  var boxW = (cv.parentElement && cv.parentElement.clientWidth) || 600;
  var S = Math.max(Math.min(boxW - 16, 460), 280);
  var dpr = window.devicePixelRatio || 1;
  cv.width = S * dpr; cv.height = S * dpr; cv.style.width = S + 'px'; cv.style.height = S + 'px';
  var c = cv.getContext('2d'); c.setTransform(dpr, 0, 0, dpr, 0, 0); c.clearRect(0, 0, S, S);
  var cx = S / 2, cy = S / 2 + 8, R = S / 2 - 46;
  var n = items.length, i;
  var ang = function (k) { return -Math.PI / 2 + k * 2 * Math.PI / n; };
  var rr = function (v) { return R * Math.max(0, Math.min(100, v)) / 100; };
  var PX = function (k, v) { return cx + rr(v) * Math.cos(ang(k)); };
  var PY = function (k, v) { return cy + rr(v) * Math.sin(ang(k)); };
  c.textAlign = 'center';
  var rings = [20, 40, 60, 80, 100];
  rings.forEach(function (rv) {
    c.beginPath();
    for (i = 0; i < n; i++) { if (i === 0) c.moveTo(PX(i, rv), PY(i, rv)); else c.lineTo(PX(i, rv), PY(i, rv)); }
    c.closePath();
    c.strokeStyle = rv === 100 ? '#66bb6a' : '#d7e8da';
    c.lineWidth = rv === 100 ? 1.5 : 1;
    c.stroke();
    c.fillStyle = '#7da88c'; c.font = '10px sans-serif';
    c.fillText(String(rv), cx + 4, cy - rr(rv) + 3);
  });
  for (i = 0; i < n; i++) {
    c.beginPath(); c.moveTo(cx, cy); c.lineTo(PX(i, 100), PY(i, 100));
    c.strokeStyle = '#d7e8da'; c.lineWidth = 1; c.stroke();
  }
  c.beginPath();
  items.forEach(function (s, k) { if (k === 0) c.moveTo(PX(k, s.value), PY(k, s.value)); else c.lineTo(PX(k, s.value), PY(k, s.value)); });
  c.closePath();
  c.fillStyle = 'rgba(102,187,106,.28)'; c.fill();
  c.strokeStyle = '#2e7d32'; c.lineWidth = 2; c.stroke();
  c.font = '11px sans-serif';
  items.forEach(function (s, k) {
    var px = PX(k, s.value), py = PY(k, s.value);
    c.beginPath(); c.arc(px, py, 3.5, 0, 7);
    c.fillStyle = s.value < 60 ? '#d81b60' : '#2e7d32'; c.fill();
    var lx = cx + (R + 18) * Math.cos(ang(k)), ly = cy + (R + 18) * Math.sin(ang(k));
    c.fillStyle = '#1f3a2d';
    c.fillText(String(s.name), lx, ly + 4);
  });
}

function trendSlipSVG() {
  var pts = (lastTrend || []).filter(function (p) { return (p.average !== null && p.average !== undefined) || (p.rank !== null && p.rank !== undefined); });
  if (!pts.length) return '';
  var W = 700, H = 300, padL = 56, padR = 48, padT = 20, padB = 48;
  var avgs = pts.map(function (p) { return p.average; }).filter(function (v) { return v !== null && v !== undefined; });
  var ranks = pts.map(function (p) { return p.rank; }).filter(function (v) { return v !== null && v !== undefined; });
  var aMin = avgs.length ? Math.min.apply(null, avgs) : 0;
  var aMax = avgs.length ? Math.max.apply(null, avgs) : 100;
  if (aMin === aMax) { aMin -= 5; aMax += 5; }
  var rMax = ranks.length ? Math.max.apply(null, ranks.concat([1])) : 1;
  var X = function (i) { return padL + (W - padL - padR) * (pts.length === 1 ? 0.5 : i / (pts.length - 1)); };
  var YA = function (v) { return padT + (H - padT - padB) * (1 - (v - aMin) / (aMax - aMin)); };
  var YR = function (v) { return padT + (H - padT - padB) * ((v - 1) / ((rMax - 1) || 1)); };
  var F = function (n) { return Math.round(Number(n) * 100) / 100; };
  var s = '';
  for (var g = 0; g <= 4; g++) {
    var y = F(padT + (H - padT - padB) * g / 4);
    s += '<line x1="' + padL + '" y1="' + y + '" x2="' + (W - padR) + '" y2="' + y + '" stroke="#cfe6d4"/>';
    s += '<line x1="' + (padL - 6) + '" y1="' + y + '" x2="' + padL + '" y2="' + y + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<line x1="' + (W - padR) + '" y1="' + y + '" x2="' + (W - padR + 6) + '" y2="' + y + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<text x="2" y="' + F(y + 4) + '" font-size="12" fill="#1565c0">' + F(aMax - (aMax - aMin) * g / 4).toFixed(1) + '</text>';
    s += '<text x="' + (W - padR + 9) + '" y="' + F(y + 4) + '" font-size="12" fill="#d81b60">' + Math.round(1 + (rMax - 1) * g / 4) + '</text>';
  }
  s += '<line x1="' + padL + '" y1="' + (padT - 8) + '" x2="' + padL + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  s += '<line x1="' + (W - padR) + '" y1="' + (padT - 8) + '" x2="' + (W - padR) + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  s += '<line x1="' + padL + '" y1="' + (H - padB) + '" x2="' + (W - padR + 8) + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  pts.forEach(function (p, i) {
    var x = F(X(i));
    s += '<line x1="' + x + '" y1="' + (H - padB) + '" x2="' + x + '" y2="' + (H - padB + 6) + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<text x="' + x + '" y="' + (H - padB + 24) + '" font-size="13" fill="#1f3a2d" text-anchor="middle">' + (p.exam_name || '') + '</text>';
  });
  s += '<text x="' + (padL + 8) + '" y="14" font-size="12" font-weight="bold" fill="#1565c0">● 平均</text>';
  s += '<text x="' + (padL + 72) + '" y="14" font-size="12" font-weight="bold" fill="#d81b60">● 班名次</text>';
  var path = function (key, Y, color) {
    var dd = '', dots = '';
    pts.forEach(function (p, i) {
      var v = p[key];
      if (v === null || v === undefined) return;
      var x = F(X(i)), y = F(Y(v));
      dd += (dd ? 'L' : 'M') + x + ' ' + y;
      dots += '<circle cx="' + x + '" cy="' + y + '" r="4.5" fill="' + color + '"/>';
    });
    return dd ? '<path d="' + dd + '" fill="none" stroke="' + color + '" stroke-width="3"/>' + dots : '';
  };
  if (avgs.length) s += path('average', YA, '#1565c0');
  if (ranks.length) s += path('rank', YR, '#d81b60');
  return '<svg class="trend" viewBox="0 0 ' + W + ' ' + H + '" xmlns="http://www.w3.org/2000/svg">' + s + '</svg>';
}

async function gridPaste(e, origin) {
  var text = (e.clipboardData || window.clipboardData).getData('text');
  if (!text || (text.indexOf(TAB) < 0 && text.indexOf(LF) < 0 && text.indexOf(CR) < 0)) return;
  e.preventDefault();
  var flat = text.split(CR).join(LF);
  var rows = flat.split(LF).filter(function (r) { return r.length; });
  var matrix = rows.map(function (r) { return r.split(TAB); });
  var ri0 = parseInt(origin.dataset.ri, 10), ci0 = parseInt(origin.dataset.ci, 10);
  var cells = [];
  matrix.forEach(function (row, di) {
    row.forEach(function (val, dj) {
      var ri = ri0 + di, ci = ci0 + dj;
      if (ri >= G.students.length || ci >= G.subjects.length) return;
      if (isDerivedSub(G.subjects[ci])) return;
      var v = String(val).trim();
      if (v === '') return;
      cells.push({ ri: ri, ci: ci, stu: G.students[ri].id, sub: G.subjects[ci].id, score: v });
    });
  });
  if (!cells.length) return;
  if (!confirm('要貼上 ' + cells.length + ' 格嗎？')) return;
  var gb = document.getElementById('gb');
  cells.forEach(function (c) {
    var inp = gb.rows[c.ri].cells[c.ci + 2].querySelector('input');
    if (inp) inp.value = c.score;
  });
  var payload = { tpw: T_PW, exam: G.exam, cells: cells.map(function (c) { return { student_id: c.stu, subject_id: c.sub, score: isNaN(Number(c.score)) ? c.score : Number(c.score) }; }) };
  try {
    var d = await api('score_bulk', payload);
    var msg = d.success ? ('✅ 已寫入 ' + d.updated + ' 格') : ('❌ ' + (d.message || '失敗'));
    if (d.errors && d.errors.length) msg += LF + '錯誤：' + LF + d.errors.join(LF);
    alert(msg);
  } catch (err) { alert('⚠ ' + err.message); }
  loadGrid();
}

async function applyBulk() {
  var info = document.getElementById('bulk-info'), err = document.getElementById('bulk-err');
  info.textContent = ''; err.style.display = 'none';
  var raw = document.getElementById('bulk').value.split(CR).join('').trim();
  if (!raw) { err.textContent = '請先貼上內容'; err.style.display = 'block'; return; }
  var lines = raw.split(LF).map(function (l) { return l.split(TAB).map(function (c) { return String(c).trim(); }); });
  var header = lines[0];
  var seatIdx = -1, nameIdx = -1;
  header.forEach(function (h, i) {
    if (seatIdx < 0 && (h.indexOf('座號') >= 0 || h === '座' || h.toLowerCase() === 'seat')) seatIdx = i;
    if (nameIdx < 0 && (h.indexOf('姓名') >= 0 || h.indexOf('名字') >= 0 || h.toLowerCase() === 'name')) nameIdx = i;
  });
  if (seatIdx < 0) seatIdx = 0;
  var subByName = {};
  (G._subs || []).forEach(function (s) { subByName[s.name] = s.id; subByName[s.code] = s.id; });
  var subById = {};
  (G._subs || []).forEach(function (s) { subById[s.id] = s; });
  var colSub = [];
  header.forEach(function (h, ci) {
    if (ci === seatIdx || ci === nameIdx) return;
    if (subByName[h] && !isDerivedSub(subById[subByName[h]])) colSub.push({ ci: ci, sid: subByName[h], name: h });
  });
  if (!colSub.length) { err.textContent = '❌ 表頭找不到對應科目。系統科目：' + (G._subs || []).map(function (s) { return s.name; }).join('、'); err.style.display = 'block'; return; }
  var seatMap = {};
  G.students.forEach(function (s) { seatMap[String(s.seat).trim()] = s.id; });
  var rows = [], warn = '';
  for (var i = 1; i < lines.length; i++) {
    var cols = lines[i];
    if (!cols.length) continue;
    var seat = String(cols[seatIdx] || '').trim();
    if (!seat) continue;
    var sid = seatMap[seat];
    if (!sid) { warn += '⚠ 第' + (i + 1) + '列座號 ' + seat + ' 不在名單，略過' + LF; continue; }
    var scores = {};
    colSub.forEach(function (cs) {
      var v = String(cols[cs.ci] || '').trim();
      if (v !== '') scores[cs.sid] = v;
    });
    rows.push({ student_id: sid, scores: scores });
  }
  if (!rows.length) { err.textContent = '❌ 無有效列'; err.style.display = 'block'; return; }
  info.textContent = '辨識 ' + rows.length + ' 人 × ' + colSub.length + ' 科，寫入中…';
  try {
    var d = await api('score_bulk', { tpw: T_PW, exam: G.exam, rows: rows });
    if (!d.success) { err.textContent = '❌ ' + (d.message || '失敗'); err.style.display = 'block'; return; }
    info.textContent = warn + '✅ 已寫入 ' + d.updated + ' 格';
    loadGrid();
  } catch (e) { err.textContent = '⚠ ' + e.message; err.style.display = 'block'; }
}
function buildStats() {
  var pure = G.subjects.filter(function (s) { return !isDerivedSub(s); });
  var kids = G.students.filter(function (st) { return !st.excluded; });
  return pure.map(function (sub) {
    var vals = kids.map(function (st) { return G.map[st.id + '_' + sub.id]; }).filter(function (v) { return v !== null && v !== undefined && v !== '' && !isNaN(Number(v)); }).map(Number);
    if (!vals.length) return { name: stripStar(sub.name), n: 0 };
    var pass = vals.filter(function (v) { return v >= 60; }).length;
    var sum = vals.reduce(function (a, b) { return a + b; }, 0);
    return { name: stripStar(sub.name), n: vals.length, avg: Math.round(sum / vals.length * 100) / 100, max: Math.max.apply(null, vals), min: Math.min.apply(null, vals), rate: Math.round(pass / vals.length * 1000) / 10, g90: vals.filter(function (v) { return v >= 90; }).length, g80: vals.filter(function (v) { return v >= 80 && v < 90; }).length, g70: vals.filter(function (v) { return v >= 70 && v < 80; }).length, g60: vals.filter(function (v) { return v >= 60 && v < 70; }).length, g0: vals.filter(function (v) { return v < 60; }).length };
  });
}

function escHtml(s) {
  return String(s == null ? '' : s).split('&').join('&amp;').split('<').join('&lt;').split('>').join('&gt;').split('"').join('&quot;');
}

// 伺服器字串注入前端 JS 單引號字串時的安全跳脫（引號/反斜線/換行/<\/script>）
function jsStr(s) {
  return String(s == null ? '' : s).replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/\n/g, '\\n').replace(/\r/g, '').replace(/<\//g, '<\\/');
}

function fmt2(v) {
  if (v === null || v === undefined || v === '') return '—';
  var n = Number(v);
  if (!isFinite(n)) return String(v);
  return String(Math.round(n * 100) / 100);
}

  var W = 700, H = 300, padL = 56, padR = 48, padT = 20, padB = 48;
  var avgs = use.map(function (p) { return p.average; }).filter(function (v) { return v !== null && v !== undefined; });
  var ranks = use.map(function (p) { return p.rank; }).filter(function (v) { return v !== null && v !== undefined; });
  var aMin = avgs.length ? Math.min.apply(null, avgs) : 0;
  var aMax = avgs.length ? Math.max.apply(null, avgs) : 100;
  if (aMin === aMax) { aMin -= 5; aMax += 5; }
  var rMax = ranks.length ? Math.max.apply(null, ranks.concat([1])) : 1;
  var X = function (i) { return padL + (W - padL - padR) * (use.length === 1 ? 0.5 : i / (use.length - 1)); };
  var YA = function (v) { return padT + (H - padT - padB) * (1 - (v - aMin) / (aMax - aMin)); };
  var YR = function (v) { return padT + (H - padT - padB) * ((v - 1) / ((rMax - 1) || 1)); };
  var F = function (n) { return Math.round(Number(n) * 100) / 100; };
  var s = '';
  for (var g = 0; g <= 4; g++) {
    var y = F(padT + (H - padT - padB) * g / 4);
    s += '<line x1="' + padL + '" y1="' + y + '" x2="' + (W - padR) + '" y2="' + y + '" stroke="#cfe6d4"/>';
    s += '<line x1="' + (padL - 6) + '" y1="' + y + '" x2="' + padL + '" y2="' + y + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<line x1="' + (W - padR) + '" y1="' + y + '" x2="' + (W - padR + 6) + '" y2="' + y + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<text x="2" y="' + F(y + 4) + '" font-size="12" fill="#1565c0">' + F(aMax - (aMax - aMin) * g / 4).toFixed(1) + '</text>';
    s += '<text x="' + (W - padR + 9) + '" y="' + F(y + 4) + '" font-size="12" fill="#d81b60">' + Math.round(1 + (rMax - 1) * g / 4) + '</text>';
  }
  s += '<line x1="' + padL + '" y1="' + (padT - 8) + '" x2="' + padL + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  s += '<line x1="' + (W - padR) + '" y1="' + (padT - 8) + '" x2="' + (W - padR) + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  s += '<line x1="' + padL + '" y1="' + (H - padB) + '" x2="' + (W - padR + 8) + '" y2="' + (H - padB) + '" stroke="#1f3a2d" stroke-width="2"/>';
  use.forEach(function (p, i) {
    var x = F(X(i));
    s += '<line x1="' + x + '" y1="' + (H - padB) + '" x2="' + x + '" y2="' + (H - padB + 6) + '" stroke="#1f3a2d" stroke-width="2"/>';
    s += '<text x="' + x + '" y="' + (H - padB + 24) + '" font-size="13" fill="#1f3a2d" text-anchor="middle">' + escHtml(p.exam_name || '') + '</text>';
  });
  s += '<text x="' + (padL + 8) + '" y="14" font-size="12" font-weight="bold" fill="#1565c0">● 平均</text>';
  s += '<text x="' + (padL + 72) + '" y="14" font-size="12" font-weight="bold" fill="#d81b60">● 班名次</text>';
  var path = function (key, Y, color) {
    var dd = '', dots = '';
    use.forEach(function (p, i) {
      var v = p[key];
      if (v === null || v === undefined) return;
      var x = F(X(i)), y = F(Y(v));
      dd += (dd ? 'L' : 'M') + x + ' ' + y;
      dots += '<circle cx="' + x + '" cy="' + y + '" r="4.5" fill="' + color + '"/>';
    });
    return dd ? '<path d="' + dd + '" fill="none" stroke="' + color + '" stroke-width="3"/>' + dots : '';
  };
  if (avgs.length) s += path('average', YA, '#1565c0');
  if (ranks.length) s += path('rank', YR, '#d81b60');
  return '<svg class="trend" viewBox="0 0 ' + W + ' ' + H + '" xmlns="http://www.w3.org/2000/svg">' + s + '</svg>';
}

function printReportPage(examName, pw) {
  if (!verifyTeacher(pw)) {
    return HtmlService.createHtmlOutput('<body><p>管理密碼錯誤，無法列印。</p></body>').setTitle('列印失敗');
  }
  var g = hGrid(examName);
  if (!g.success) {
    return HtmlService.createHtmlOutput('<body><p>' + escHtml(g.message) + '</p></body>').setTitle('列印失敗');
  }
  var isAvg = function (s) { return (s.code || '').indexOf('平均') >= 0 || (s.name || '').indexOf('平均') >= 0; };
  var isDer = function (s) { return DERIVED_RE.test(s.code || '') || DERIVED_RE.test(s.name || ''); };
  var fmt = function (sub, v) {
    if (v === null || v === undefined || v === '') return '';
    if (isAvg(sub)) { var n = Number(v); return isFinite(n) ? String(Math.round(n * 100) / 100) : String(v); }
    return String(v);
  };
  var head = ['座號', '姓名'].concat(g.subjects.map(function (s) { return stripWeightSuffix(String(s.name || '')); }));
  var html = '<h2>' + escHtml(getSystemTitle()) + '總表 — ' + escHtml(examName) + '</h2>';
  var t = new Date();
  html += '<div class="rp-sub">列印日期：' + t.getFullYear() + '/' + (t.getMonth() + 1) + '/' + t.getDate() + '　共 ' + g.students.length + ' 人（未滿 60 分紅字標示）</div>';
  html += '<table><thead><tr>';
  head.forEach(function (h) { html += '<th>' + escHtml(h) + '</th>'; });
  html += '</tr></thead><tbody>';
  g.students.forEach(function (st) {
    html += '<tr' + (st.excluded ? ' class="test"' : '') + '><td>' + escHtml(st.seat == null ? '' : st.seat) + '</td><td class="name">' + escHtml(st.name || '') + (st.excluded ? '（測）' : '') + '</td>';
    g.subjects.forEach(function (sub) {
      var v = g.scores[st.id + '_' + sub.id];
      var fail = !isDer(sub) && !isAvg(sub) && v !== null && v !== undefined && v !== '' && Number(v) < 60;
      html += '<td' + (fail ? ' class="fail"' : '') + '>' + escHtml(fmt(sub, v)) + '</td>';
    });
    html += '</tr>';
  });
  html += '</tbody></table>';
  var pure = g.subjects.filter(function (s) { return !isDer(s); });
  var kids = g.students.filter(function (st) { return !st.excluded; });
  var stats = pure.map(function (sub) {
    var vals = [];
    kids.forEach(function (st) {
      var v = g.scores[st.id + '_' + sub.id];
      if (v !== null && v !== undefined && v !== '' && !isNaN(Number(v))) vals.push(Number(v));
    });
    if (!vals.length) return { name: stripWeightSuffix(sub.name), n: 0 };
    var pass = vals.filter(function (v) { return v >= 60; }).length;
    var sum = vals.reduce(function (a, b) { return a + b; }, 0);
    return { name: stripWeightSuffix(sub.name), n: vals.length, avg: Math.round(sum / vals.length * 100) / 100, max: Math.max.apply(null, vals), min: Math.min.apply(null, vals), rate: Math.round(pass / vals.length * 1000) / 10, g90: vals.filter(function (v) { return v >= 90; }).length, g80: vals.filter(function (v) { return v >= 80 && v < 90; }).length, g70: vals.filter(function (v) { return v >= 70 && v < 80; }).length, g60: vals.filter(function (v) { return v >= 60 && v < 70; }).length, g0: vals.filter(function (v) { return v < 60; }).length };
  });
  if (stats.length) html += '<h3>全班統計</h3>' + statsTableServer(stats);
  html += '<div class="rp-foot">※ 總分、平均、排名、前名次、進退步由系統自動計算。</div>';
  var doc = '<!DOCTYPE html><html lang="zh-TW"><head><meta charset="UTF-8"><title>成績總表</title>'
    + '<style>' + REPORT_CSS + '</style></head><body onload="window.print()">' + html + '</body></html>';
  return HtmlService.createHtmlOutput(doc).setTitle('成績總表');
}

// 全班個人成績單批次（一次讀取、逐生組裝，避免逐人多次讀表超時）
// ============================================================
function hClassSlips(examName) {
  var exam = String(examName || '').trim();
  var sh = getSS().getSheetByName(exam);
  if (!sh) return { success: false, message: '找不到考試：' + exam };
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return { success: false, message: '工作表無資料' };
  var headers = data[0].map(function (x) { return String(x == null ? '' : x).trim(); });
  var seatCol = findSeatCol(headers), nameCol = findNameCol(headers);
  var rowBySeat = {};
  for (var i = 1; i < data.length; i++) {
    var k = normSeat(data[i][seatCol]);
    if (k && String(data[i][seatCol]).trim() && rowBySeat[k] == null) rowBySeat[k] = i;
  }
  var pureIdx = [];
  headers.forEach(function (h, idx) {
    if (idx === seatCol || idx === nameCol || !h) return;
    if (isPureSubject(h)) pureIdx.push(idx);
  });
  var rankFor = function (pi) {
    var subj = stripWeightSuffix(headers[pi]);
    for (var j = 0; j < headers.length; j++) {
      var h = headers[j];
      if (!/排名|名次/.test(h)) continue;
      if (h === '班名次' || h === '前名次') continue;
      var base = h.replace(/排名|名次/g, '');
      if (!base) continue;
      if (subj.indexOf(base) === 0) return j;
    }
    return -1;
  };
  var rankOf = {};
  pureIdx.forEach(function (pi) { rankOf[pi] = rankFor(pi); });
  var avgIdx = -1;
  for (var a = 0; a < headers.length; a++) {
    if (headers[a].indexOf('平均') >= 0) { avgIdx = a; break; }
  }
  var summaryIdx = [];
  headers.forEach(function (h, idx) {
    if (idx === seatCol || idx === nameCol || !h) return;
    if (isPureSubject(h)) return;
    if (/排名|名次/.test(h) && h !== '班名次' && h !== '前名次') return;
    summaryIdx.push(idx);
  });
  var cellOf = function (ri, ci) {
    if (ri == null) return null;
    var v = data[ri][ci];
    return (v === '' || v == null) ? null : v;
  };
  var accounts = readAccounts();
  accounts.sort(function (x, y) {
    var a = parseFloat(x.seat), b = parseFloat(y.seat);
    if (!isNaN(a) && !isNaN(b)) return a - b;
    return String(x.seat) < String(y.seat) ? -1 : 1;
  });
  // 趨勢：每張考卷只讀一次
  var exams = orderedExamNames();
  var trendBySeat = {};
  exams.forEach(function (n) {
    var s2 = getSS().getSheetByName(n);
    if (!s2) return;
    var dd = s2.getDataRange().getValues();
    if (dd.length < 2) return;
    var hh = dd[0].map(function (x) { return String(x == null ? '' : x).trim(); });
    var sc = findSeatCol(hh);
    var ai = -1, ci = -1;
    for (var j = 0; j < hh.length; j++) {
      if (ai < 0 && hh[j].indexOf('平均') >= 0) ai = j;
      if (hh[j] === '班名次') ci = j;
    }
    for (var r = 1; r < dd.length; r++) {
      if (!String(dd[r][sc] == null ? '' : dd[r][sc]).trim()) continue;
      var kk = normSeat(dd[r][sc]);
      if (!trendBySeat[kk]) trendBySeat[kk] = [];
      trendBySeat[kk].push({ exam_name: n, average: ai >= 0 ? toNum(dd[r][ai]) : null, rank: ci >= 0 ? toNum(dd[r][ci]) : null });
    }
  });
  var students = accounts.map(function (ac) {
    var ri = rowBySeat[normSeat(ac.seat)];
    var subjects = pureIdx.map(function (pi) {
      var rj = rankOf[pi];
      return { name: stripWeightSuffix(headers[pi]), score: cellOf(ri, pi), rank: rj >= 0 ? cellOf(ri, rj) : null };
    });
    var summary = summaryIdx.map(function (si) { return { name: headers[si], value: cellOf(ri, si) }; });
    var offAvg = avgIdx >= 0 ? toNum(cellOf(ri, avgIdx)) : null;
    var sum = 0, nn = 0;
    pureIdx.forEach(function (pi) {
      var v = toNum(cellOf(ri, pi));
      if (v !== null) { sum += v; nn++; }
    });
    var avg = nn ? Math.round(sum / nn * 100) / 100 : null;
    var tb = trendBySeat[normSeat(ac.seat)] || [];
    return { account: ac.account, seat: ac.seat, name: ac.name + (ac.excluded ? '（測）' : ''), classNo: ac.classNo, excluded: ac.excluded, subjects: subjects, summary: summary, avg: (offAvg !== null && offAvg !== undefined) ? offAvg : avg, trend: tb.filter(function (p) { return p.average != null || p.rank != null; }), hasRow: ri != null };
  });
  var stats = [];
  try {
    var st = hStudentStats(exam);
    if (st.success) stats = st.stats || [];
  } catch (e) {}
  return { success: true, exam: exam, students: students, stats: stats };
}
